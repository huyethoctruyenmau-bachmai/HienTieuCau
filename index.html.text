function bindAutocomplete(options){
  const key=options.textId;
  if(AutocompleteRegistry[key]){AutocompleteRegistry[key].update(options);return AutocompleteRegistry[key].api}
  const input=Ui.el(options.textId),hidden=Ui.el(options.hiddenId),box=Ui.el(options.suggestionsId);if(!input||!hidden||!box)return null;
  const state={options:{...options},items:[]};
  const label=x=>state.options.labelFn(x);
  const sub=x=>state.options.subFn?.(x)||'';
  const searchText=x=>state.options.searchFn?state.options.searchFn(x):(label(x)+' '+sub(x));
  let clearBtn=null;
  const syncClear=()=>{if(clearBtn)clearBtn.classList.toggle('hidden',!input.value)};
  const selectItem=x=>{if(!x)return false;hidden.value=x.code??x.maNhanVien??'';input.value=label(x);box.classList.add('hidden');syncClear();state.options.onSelect?.(x);return true};
  const render=()=>{
    const q=Ui.norm(input.value);
    state.items=(state.options.getItems?.()||[]).filter(x=>!q||Ui.norm(searchText(x)).includes(q)).slice(0,25);
    if(!state.items.length){box.classList.add('hidden');box.innerHTML='';return}
    box.innerHTML=state.items.map((x,i)=>`<div class="suggestion" data-i="${i}"><div class="suggestion-main">${Ui.esc(label(x))}</div>${state.options.subFn?`<div class="suggestion-sub">${Ui.esc(sub(x))}</div>`:''}</div>`).join('');
    box.classList.remove('hidden');
  };
  input.addEventListener('input',()=>{hidden.value='';syncClear();render()});
  input.addEventListener('focus',render);
  input.addEventListener('keydown',e=>{if(e.key==='Enter'&&!box.classList.contains('hidden')&&state.items.length){e.preventDefault();e.stopPropagation();selectItem(state.items[0]);Ui.focusNext(input)}});

  box.addEventListener('mousedown',e=>{

    selectItem(x);
  });

    clear(){hidden.value='';input.value='';box.classList.add('hidden');syncClear()},
    refresh(){render()}
  };
  AutocompleteRegistry[key]={api,update(next){state.options={...state.options,...next}}};
  return api;

    }
  },
